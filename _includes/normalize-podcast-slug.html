{%- comment -%} Normalize a podcast slug from Decap relation (full filename slug) or manual entry. Sets `normalized_podcast_slug`. Params: slug {%- endcomment -%} {%- assign normalized_podcast_slug = include.slug | strip -%} {%- if normalized_podcast_slug != '' -%} {%- assign _parts = normalized_podcast_slug | split: '-' -%} {%- if _parts.size >= 4 and _parts[0].size == 4 and _parts[1].size <= 2 and _parts[2].size <= 2 -%} {%- assign normalized_podcast_slug = _parts | slice: 3, 100 | join: '-' -%} {%- endif -%} {%- assign _md_tail = normalized_podcast_slug | slice: -3, 3 -%} {%- if _md_tail == '-md' -%} {%- assign _strip_len = normalized_podcast_slug.size | minus: 3 -%} {%- assign normalized_podcast_slug = normalized_podcast_slug | slice: 0, _strip_len -%} {%- endif -%} {%- endif -%}
