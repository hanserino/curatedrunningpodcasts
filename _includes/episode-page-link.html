{%- comment -%} Renders episode title as link when we have a site episode page, else YouTube watch URL, else plain text. Params: episode (hash), podcast_slug (optional), podcast_page_url (optional), class (optional), datetime_attr (optional) {%- endcomment -%} {%- assign _episode = include.episode -%} {%- assign _class = include.class | default: '' -%} {%- assign _audio = _episode.audio_url | default: '' | strip -%} {%- assign _yt_id = _episode.youtube_video_id | default: '' | strip -%} {%- assign _yt_url = '' -%} {%- if _yt_id != '' -%} {%- assign _yt_url = 'https://www.youtube.com/watch?v=' | append: _yt_id -%} {%- elsif _episode.episode_url and _episode.episode_url != '' -%} {%- if _episode.episode_url contains 'youtube.com' or _episode.episode_url contains 'youtu.be' -%} {%- assign _yt_url = _episode.episode_url -%} {%- endif -%} {%- endif -%} {%- assign _ep_path = _episode.episode_page_url | default: '' | strip -%} {%- if _audio != '' -%} {%- if _ep_path == '' -%} {%- assign _key = _episode.episode_slug | default: _episode.episode_key | default: '' | strip -%} {%- if _key == '' -%} {%- assign _key = _episode.episode_title | slugify -%} {%- endif -%} {%- assign _slug = include.podcast_slug | default: '' | strip -%} {%- if _slug == '' and include.podcast_page_url -%} {%- include podcast-slug-from-url.html url=include.podcast_page_url -%} {%- assign _slug = podcast_slug_from_url -%} {%- endif -%} {%- if _slug != '' and _key != '' -%} {%- assign _ep_path = '/' | append: _slug | append: '/' | append: _key | append: '/' -%} {%- endif -%} {%- endif -%} {%- else -%} {%- assign _ep_path = '' -%} {%- endif -%} {%- if _ep_path != '' -%} {{ _episode.episode_title }} {%- elsif _yt_url != '' -%} {{ _episode.episode_title }} {%- else -%} {{ _episode.episode_title }} {%- endif -%}
