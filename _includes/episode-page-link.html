{%- comment -%} Renders episode title as link when we have a site episode page, else plain text. Params: episode (hash), podcast_slug (optional), podcast_page_url (optional), class (optional), datetime_attr (optional) {%- endcomment -%} {%- assign _episode = include.episode -%} {%- assign _class = include.class | default: '' -%} {%- assign _episode_pages_enabled = site.podcast_episode_pages_enabled -%} {%- assign _ep_path = '' -%} {%- unless _episode_pages_enabled == false -%} {%- assign _ep_path = _episode.episode_page_url | default: '' | strip -%} {%- endunless -%} {%- if _ep_path == '' and _episode_pages_enabled != false -%} {%- assign _key = _episode.episode_slug | default: _episode.episode_key | default: '' | strip -%} {%- unless _episode.episode_page_url -%} {%- assign _key = _episode.episode_title | slugify -%} {%- endunless -%} {%- assign _slug = include.podcast_slug | default: '' | strip -%} {%- if _slug == '' and include.podcast_page_url -%} {%- include podcast-slug-from-url.html url=include.podcast_page_url -%} {%- assign _slug = podcast_slug_from_url -%} {%- endif -%} {%- if _slug != '' and _key != '' -%} {%- assign _ep_path = '/' | append: _slug | append: '/' | append: _key | append: '/' -%} {%- endif -%} {%- endif -%} {%- if _ep_path != '' -%} {{ _episode.episode_title }} {%- else -%} {{ _episode.episode_title }} {%- endif -%}
